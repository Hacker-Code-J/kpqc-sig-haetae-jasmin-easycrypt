\begin{frame}{\bfseries Artifact Claim}{What the Artifact Proves}

\begin{center}
\large\bfseries
The artifact gives a fully machine-checked \easycrypt{} proof that the scalar
\jasmin{} NTT code implements the intended HAETAE transform.
\end{center}

\vspace{10pt}
\begin{center}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{p{4.2cm}p{8.2cm}}
\toprule[1.2pt]
\textbf{Forward NTT} &
\ec{poly\_ntt\_jazz} returns an array representing $\mathtt{full\_ntt}(p)$,
with output bound $2^{24}$. \\
\midrule
\textbf{Inverse NTT} &
\ec{poly\_invntt\_jazz} returns an array representing
$\mathtt{array256\_mont}(\mathtt{full\_invntt}(p))$, the Montgomery-form result
expected by HAETAE arithmetic. \\
\bottomrule[1.2pt]
\end{tabular}
\end{center}

\vspace{8pt}
\begin{center}
\small The claim is functional correctness of the scalar NTT primitives, not a
full proof of the complete signature scheme.
\end{center}
\end{frame}

\begin{frame}{\bfseries Artifact Claim}{Mathematical Notation Behind the Claim}
\small
\defbox[Field, root, and decoding map]{
Let
\[
  q=64513,\qquad
  \mathbb F_q=\mathbb Z/q\mathbb Z,\qquad
  \zeta=[426]_q,\qquad
  R=[2^{32}]_q.
\]
We write $\brev_8(i)$ for \ec{br i}, where \ec{br = bsrev 8};
$p_i$ for \ec{p.[i]}; and $\zeta^e$ for
\ec{ZqRing.exp zroot e}.

For a Jasmin byte array \ec{rp}, define the decoded polynomial
\[
  D(rp)_i =
  \left[
    \operatorname{to\_sint}\bigl(\operatorname{get32}(rp,i)\bigr)
  \right]_q,
  \qquad 0\le i<256.
\]
}

\vspace{4pt}
\thmbox[Exact meaning of \ec{NTT\_Fq.poly\_repr\_bound rp p b}]{
\[
  \mathsf{Repr}_b(rp,p)
  \;\Longleftrightarrow\;
  D(rp)=p
  \;\land\;
  \forall i\in[0,256).\;
  -2^b \le \operatorname{to\_sint}(\operatorname{get32}(rp,i)) < 2^b.
\]
}
\end{frame}

%-------------------------------------------------------
\begin{frame}[fragile]{\bfseries Artifact Claim}{Forward NTT Theorem}
\begin{ecblock}[NTTEndToEnd.ec]
lemma haetae_jasmin_ntt_correct p :
  hoare [Hpoly_extract.M.poly_ntt_jazz :
    NTT_Fq.poly_repr_bound rp p 16 ==>
    NTT_Fq.poly_repr_bound res (NTTFullSpec.full_ntt p) 24].
\end{ecblock}

\vspace{4pt}

\thmbox[Mathematical formula carried by the code]{
\scriptsize
\[
  F(p)_i=
  \sum_{j=0}^{255} p_j\,\zeta^{(2\brev_8(i)+1)j},
  \qquad 0\le i<256.
\]
\[
  \forall p\in\mathbb F_q^{256}.\quad
  \bigl\{\mathsf{Repr}_{16}(rp,p)\bigr\}\;
  \ec{poly\_ntt\_jazz}(rp)\;
  \bigl\{\mathsf{Repr}_{24}(res,F(p))\bigr\}.
\]
Thus the EasyCrypt postcondition is not merely termination: it says that the
returned Jasmin memory decodes to the full HAETAE forward transform.
}
\end{frame}

%-------------------------------------------------------
\begin{frame}{\bfseries Artifact Claim}{Forward Theorem as a Formula}
\small
\defbox[Mathematical transform corresponding to \ec{NTTFullSpec.full\_ntt}]{
\[
  F(p)_i
  =
  \sum_{j=0}^{255}
    p_j\,\zeta^{(2\brev_8(i)+1)j},
  \qquad 0\le i<256.
\]
}

\vspace{5pt}
\thmbox[Hoare triple proved by \ec{haetae\_jasmin\_ntt\_correct}]{
\[
  \bigl\{\mathsf{Repr}_{16}(rp,p)\bigr\}\;
  \ec{poly\_ntt\_jazz}(rp)\;
  \bigl\{\mathsf{Repr}_{24}(res,F(p))\bigr\}.
\]
Equivalently, for every input polynomial $p\in\mathbb F_q^{256}$ represented
by the input memory, the returned Jasmin memory decodes to exactly $F(p)$ and
its signed words are bounded by $2^{24}$.
}
\end{frame}

%-------------------------------------------------------
\begin{frame}[fragile]{\bfseries Artifact Claim}{Inverse NTT Theorem}
\begin{ecblock}[NTTEndToEnd.ec]
lemma haetae_jasmin_invntt_correct p :
  hoare [Hpoly_extract.M.poly_invntt_jazz :
    NTT_Fq.poly_repr_bound rp p 16 ==>
    NTT_Fq.poly_repr_bound res
      (NTT_Fq.array256_mont (NTTFullSpec.full_invntt p)) 16].
\end{ecblock}

\vspace{4pt}

\thmbox[Mathematical formula carried by the code]{
\scriptsize
\[
  F^{-1}(p)_i =
  [256]_q^{-1}\sum_{j=0}^{255}
    p_j\,\zeta^{-((2\brev_8(j)+1)i)},\qquad
  \mathsf{Mont}(v)_i=R\,v_i.
\]
\[
  \forall p\in\mathbb F_q^{256}.\quad
  \bigl\{\mathsf{Repr}_{16}(rp,p)\bigr\}\;
  \ec{poly\_invntt\_jazz}(rp)\;
  \bigl\{\mathsf{Repr}_{16}(res,\mathsf{Mont}(F^{-1}(p)))\bigr\}.
\]
The \ec{array256\_mont} wrapper in the EasyCrypt code is exactly the
coefficientwise factor $R=[2^{32}]_q$ in this formula.
}
\end{frame}

%-------------------------------------------------------
\begin{frame}{\bfseries Artifact Claim}{Inverse Theorem as a Formula}
\small
\defbox[Mathematical transform corresponding to \ec{NTTFullSpec.full\_invntt}]{
\[
  F^{-1}(p)_i
  =
  [256]_q^{-1}
  \sum_{j=0}^{255}
    p_j\,\zeta^{-((2\brev_8(j)+1)i)},
  \qquad 0\le i<256.
\]
\[
  \ec{NTT\_Fq.array256\_mont}(v)_i = R\,v_i.
\]
}

\vspace{5pt}
\thmbox[Hoare triple proved by \ec{haetae\_jasmin\_invntt\_correct}]{
\[
  \bigl\{\mathsf{Repr}_{16}(rp,p)\bigr\}\;
  \ec{poly\_invntt\_jazz}(rp)\;
  \bigl\{\mathsf{Repr}_{16}(res,R\cdot F^{-1}(p))\bigr\}.
\]
The factor $R=[2^{32}]_q$ is the Montgomery representation factor appearing
in the EasyCrypt postcondition
\ec{array256\_mont (full\_invntt p)}.
}
\end{frame}

\begin{frame}{\bfseries Artifact Claim}{Why This Was Not Just Reusing Kyber}
\begin{columns}[T]
\column{0.48\textwidth}
\defbox[What could be reused]{
\normalsize
\begin{itemize}
  \setlength\itemsep{6pt}
  \item The high-assurance stack: \jasmin{} source, \jasmin{} extraction, and \easycrypt{} proof.
  \item The overall proof shape: refine low-level code to an imperative NTT, then connect the loop to algebra.
  \item Standard word-arithmetic and relational Hoare-logic techniques.
\end{itemize}
}

\column{0.48\textwidth}
\defbox[What had to be redone]{
\normalsize
\begin{itemize}
  \setlength\itemsep{6pt}
  \item HAETAE works with $q=64513$ and primitive root $\zeta_0=426$.
  \item The intended transform is the full 256-point spectral NTT over $X^{256}+1$.
  \item The Kyber-style ring-level NTT was not the final HAETAE specification, so a new full spectral specification and algebra bridge were needed.
\end{itemize}
}
\end{columns}

\vspace{6pt}
\begin{center}
\small The artifact closes that HAETAE-specific bridge inside \ec{NTTFullAlgebra.ec}.
\end{center}
\end{frame}

\begin{frame}{\bfseries Formulation Correspondence}{Forward Butterfly Lemma}
\footnotesize
Let
\[
  P_{\ell,s,b}(p)
  =
  \sum_{u=0}^{\ell-1}
    \zeta_0^{(2s+1)\brev_8(u)}
    p_{\brev_8(b\ell+u)}.
\]
For stage $k$, block $ks$, offset $off$, define
\[
\begin{aligned}
  a &= ks\,2^{8-k}+off,
  &
  b &= a+2^{7-k},\\
  z &= \ec{zetas}[2^k+ks]
     =\zeta_0^{(2\brev_k(ks)+1)2^{7-k}}.
\end{aligned}
\]

\vspace{4pt}
The code assignment
\[
  t=z r_b,\qquad r_b\gets r_a-t,\qquad r_a\gets r_a+t
\]
is the mathematical update
\[
  r'_a=r_a+zr_b,\qquad r'_b=r_a-zr_b,\qquad
  r'_x=r_x\;(x\notin\{a,b\}).
\]

\vspace{4pt}
The EasyCrypt lemma \ec{forward\_nat\_pair\_refines} proves:
\[
\begin{gathered}
 r_a=P_{2^k,\brev_k(ks),\,2\brev_{7-k}(off)}(p),\quad
 r_b=P_{2^k,\brev_k(ks),\,2\brev_{7-k}(off)+1}(p)\\
\Longrightarrow\quad
 r'_a=P_{2^{k+1},\brev_k(ks),\,\brev_{7-k}(off)}(p),\quad
 r'_b=P_{2^{k+1},2^k+\brev_k(ks),\,\brev_{7-k}(off)}(p).
\end{gathered}
\]
\end{frame}

%-------------------------------------------------------
\begin{frame}{\bfseries Formulation Correspondence}{Forward Butterfly Explained}
\small
\defbox[Meaning of $P_{\ell,s,b}$]{
$P_{\ell,s,b}(p)$ is a partial NTT sum.  It looks only at the block numbered
$b$, of length $\ell$, and evaluates that block at the exponent pattern
determined by $s$:
\[
  P_{\ell,s,b}(p)
  =
  \sum_{u=0}^{\ell-1}
    \zeta_0^{(2s+1)\brev_8(u)}
    p_{\brev_8(b\ell+u)}.
\]
}

\vspace{6pt}
\thmbox[Why the butterfly is correct]{
Before the butterfly, the two touched cells contain two length-$2^k$ partial
sums.  Multiplying the second one by
\[
  z=\zeta_0^{(2\brev_k(ks)+1)2^{7-k}}
\]
aligns its exponents with the next stage.  The two assignments
$r'_a=r_a+zr_b$ and $r'_b=r_a-zr_b$ are therefore exactly the two
length-$2^{k+1}$ partial sums required by the next stage.
}
\end{frame}

\begin{frame}{\bfseries Formulation Correspondence}{Forward Stage Invariant}
\small
\defbox[Stage predicate]{
\[
  \mathsf{Stage}_\ell(r,p)
  \Longleftrightarrow
  \forall s,b.\;
  0\le s<\ell \land 0\le b<256/\ell
  \Rightarrow
  r_{\brev_8(b\ell+s)}=P_{\ell,s,b}(p).
\]
In EasyCrypt this is \ec{full\_stage\_spec r p len}.
}

\vspace{8pt}
\thmbox[Mathematical content of the forward proof]{
\[
\begin{array}{rcll}
\mathsf{Stage}_1(p,p) &&& \ec{full\_stage1\_init},\\[2pt]
\mathsf{Stage}_{2^k}(r,p)
  &\Longrightarrow&
\mathsf{Stage}_{2^{k+1}}(r',p)
& \ec{fwd\_stage\_progress\_finish},\\[2pt]
\mathsf{Stage}_{256}(r,p)
  &\Longrightarrow&
r=\ec{full\_ntt}(p)
& \ec{full\_stage256\_imp}.
\end{array}
\]
}

\vspace{6pt}
\begin{center}
Thus the nested loops of \ec{NTT.ntt} are a checked induction over
$\ell=1,2,4,\ldots,256$.
\end{center}
\end{frame}

%-------------------------------------------------------
\begin{frame}{\bfseries Formulation Correspondence}{Forward Stage Invariant Explained}
\small
\defbox[What the invariant buys us]{
The statement
\[
  \mathsf{Stage}_\ell(r,p)
\]
does not merely say that a few cells are correct.  It says that every cell of
the current array is already the correct partial transform for block length
$\ell$.
}

\vspace{6pt}
\thmbox[How the loop proof uses it]{
The proof starts with $\ell=1$, where each partial sum is just one input
coefficient.  Each completed forward stage doubles the length:
\[
  \mathsf{Stage}_{2^k}(r,p)
  \Longrightarrow
  \mathsf{Stage}_{2^{k+1}}(r',p).
\]
After eight stages, $\ell=256$ and there is only one block.  The invariant then
collapses to the full formula for \ec{full\_ntt}.
}
\end{frame}

\begin{frame}{\bfseries Formulation Correspondence}{Inverse Butterfly Lemma}
\footnotesize
Let
\[
  Q_{\ell,s,b}(y)
  =
  \sum_{u=0}^{\ell-1}
    y_{b\ell+u}\,
    \zeta_0^{-((2\brev_8(b\ell+u)+1)s)}.
\]
For inverse stage $k$, block $ks$, offset $off$, define
\[
\begin{aligned}
  a &= ks\,2^{k+1}+off,
  &
  b &= a+2^k,\\
  z &= \ec{zetas\_inv}[256-2^{8-k}+ks]\\
    &= \zeta_0^{-((2\brev_{7-k}(ks)+1)2^k)}.
\end{aligned}
\]

\vspace{4pt}
The code assignment
\[
  t=r_a,\qquad r_a\gets t+r_b,\qquad r_b\gets z(t-r_b)
\]
is the mathematical update
\[
  r'_a=r_a+r_b,\qquad r'_b=z(r_a-r_b).
\]

\vspace{4pt}
\ec{inverse\_pair\_refines\_values} proves:
\[
\begin{gathered}
 r_a=Q_{2^k,off,2ks}(y),\quad
 r_b=Q_{2^k,off,2ks+1}(y)\\
\Longrightarrow\quad
 r'_a=Q_{2^{k+1},off,ks}(y),\quad
 r'_b=Q_{2^{k+1},2^k+off,ks}(y).
\end{gathered}
\]
\end{frame}

\begin{frame}{\bfseries Formulation Correspondence}{Inverse Butterfly Explained}
\small
\defbox[Meaning of $Q_{\ell,s,b}$]{
$Q_{\ell,s,b}(y)$ is the inverse counterpart of the forward partial sum.  It
accumulates spectral coefficients from one block and uses the inverse exponent
pattern:
\[
  Q_{\ell,s,b}(y)
  =
  \sum_{u=0}^{\ell-1}
    y_{b\ell+u}\,
    \zeta_0^{-((2\brev_8(b\ell+u)+1)s)}.
\]
}

\vspace{6pt}
\thmbox[How the inverse loop differs]{
The inverse butterfly first forms $r_a+r_b$ and $r_a-r_b$, then multiplies the
difference by the inverse-stage root
\[
  \zeta_0^{-((2\brev_{7-k}(ks)+1)2^k)}.
\]
The EasyCrypt proof shows that this merges two length-$2^k$ inverse partial
sums into two length-$2^{k+1}$ inverse partial sums.  The final loop multiplies
all coefficients by \ec{zetas\_inv[255]}$=256^{-1}$.
}
\end{frame}

\begin{frame}{\bfseries Lemma Reasoning}{Root Period and Evaluation Points}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Period facts}
\small
\[
  \zeta_0^{256} = -1,
  \qquad
  \zeta_0^{512} = 1.
\]
\ec{exp\_zroot\_512} follows by squaring the checked fact
$\zeta_0^{256}=-1$.  Then \ec{exp\_zroot\_mod512} lets the proof rewrite:
\[
  \zeta_0^{e+512n} = \zeta_0^e.
\]

\column{0.48\textwidth}
\textbf{Distinct evaluation points}
\small
The full NTT uses:
\[
  \omega_i = \zeta_0^{2\brev_8(i)+1}.
\]
If $\omega_i=\omega_j$, then:
\[
  \zeta_0^{2(\brev_8(i)-\brev_8(j))}=1.
\]
Because $\zeta_0$ has period 512, this forces
$256 \mid \brev_8(i)-\brev_8(j)$.  Both values lie in $[0,256)$, so
$\brev_8(i)=\brev_8(j)$, hence $i=j$.
\end{columns}
\end{frame}

%-------------------------------------------------------
\begin{frame}{\bfseries Lemma Reasoning}{Bit-Reversal Stage Index}
\thmbox[\ec{bsrev\_stage\_index}]{
\normalsize
For $0\le k<8$ and $0\le ks<2^k$:
\[
  \brev_8(2^k+ks)
  =
  (2\brev_k(ks)+1)\,2^{7-k}.
\]
}

\vspace{6pt}
\begin{center}
\small\renewcommand{\arraystretch}{1.35}
\begin{tabular}{p{4.1cm}p{8.0cm}}
\toprule[1.2pt]
\textbf{Bit pattern before reversal} &
$2^k+ks$ has bit $k$ set to $1$, lower $k$ bits equal to $ks$, and all higher bits zero. \\
\midrule
\textbf{After 8-bit reversal} &
The set bit moves to position $7-k$, the lower $k$ bits become
$\brev_k(ks)$ above it, and the lower $7-k$ bits become zero. \\
\midrule
\textbf{Integer value} &
$2^{7-k} + \brev_k(ks)\,2^{8-k}
= (2\brev_k(ks)+1)2^{7-k}$. \\
\bottomrule[1.2pt]
\end{tabular}
\end{center}

\vspace{4pt}
\begin{center}
\small This is the bridge from loop counter \ec{2\^{}k + ks} to the spectral
exponent used by the mathematical NTT.
\end{center}
\end{frame}

%-------------------------------------------------------
\begin{frame}{\bfseries Lemma Reasoning}{Twiddle Table and Montgomery Cancellation}
\begin{columns}[T]
\column{0.49\textwidth}
\textbf{\ec{zetas\_stage}}
\small
The table is defined by bit-reversed exponents:
\[
  \mathtt{zetas}[t]=\zeta_0^{\brev_8(t)}.
\]
Substitute $t=2^k+ks$ and use \ec{bsrev\_stage\_index}:
\[
  \mathtt{zetas}[2^k+ks]
  =
  \zeta_0^{(2\brev_k(ks)+1)2^{7-k}}.
\]

\column{0.49\textwidth}
\textbf{Montgomery multiplication}
\small
The Jasmin table stores the Montgomery value
$\zeta R \bmod q$.  The reducer satisfies:
\[
  \SREDC(x) \equiv xR^{-1} \pmod q.
\]
Therefore the butterfly multiplication computes:
\[
  \SREDC((\zeta R)\,a)
  \equiv
  \zeta a \pmod q.
\]
The stored $R$ and the reducer's $R^{-1}$ cancel exactly.
\end{columns}
\end{frame}

%-------------------------------------------------------
\begin{frame}{\bfseries Lemma Reasoning}{Forward Butterfly Stage Lemma}
\begin{center}
\small
The key local step is represented by \ec{fwd\_stage\_progress\_step} and then
lifted through loop invariants to \ec{ntt\_full\_ntt}.
\end{center}

\vspace{4pt}
\begin{columns}[T]
\column{0.49\textwidth}
\textbf{Program update}
\small
For one pair $(a,b)$ and twiddle $w$:
\[
\begin{aligned}
  t &= w\cdot b,\\
  a' &= a+t,\\
  b' &= a-t.
\end{aligned}
\]
The proof also shows the two array addresses are distinct, so the in-place
updates affect only that pair.

\column{0.49\textwidth}
\textbf{Mathematical meaning}
\small
At stage $k$, $a$ and $b$ are two partial sums over complementary halves of
one residue class.  The twiddle from \ec{zetas\_stage} supplies exactly the
missing factor needed to merge the halves:
\[
  \text{partial}_k^{\mathrm{low}}
  + w\,\text{partial}_k^{\mathrm{high}}
  =
  \text{partial}_{k+1}.
\]
The lower output uses the same split with a sign change.
\end{columns}
\end{frame}

%-------------------------------------------------------
\begin{frame}{\bfseries Lemma Reasoning}{Inverse Loop and Final Scaling}
\begin{columns}[T]
\column{0.49\textwidth}
\textbf{Reverse-stage reasoning}
\small
\ec{partial\_invntt\_split\_low} and
\ec{partial\_invntt\_split\_high} split the inverse spectral sum into the two
halves touched by one reverse butterfly.  \ec{zetas\_inv\_stage} supplies the
inverse twiddle factor, so one program pair reconstructs the next inverse
partial sum.

\vspace{4pt}
The stage lemmas are threaded through the outer invariant until the loop exit
postcondition, ending at \ec{inv\_outer\_exit\_post}.

\column{0.49\textwidth}
\textbf{Post-processing}
\small
The inverse routine has a final scaling pass.  The proof uses
\ec{zetas\_inv\_255} and \ec{inv\_post\_exit\_full} to show that this pass
multiplies by the required inverse factor:
\[
  256^{-1}.
\]
The end-to-end theorem then wraps the mathematical inverse in
\ec{array256\_mont}, matching HAETAE's representation convention.
\end{columns}
\end{frame}
