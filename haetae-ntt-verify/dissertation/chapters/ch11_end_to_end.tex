\section{Main Correctness Theorems}

\subsection{Forward NTT Correctness}

\begin{theorem}[Jasmin NTT correctness]
  \label{thm:main_ntt}
  Let $\texttt{rp} : \ec{W32.t}^{256}$ be a word array encoding a polynomial
  with 16-bit signed coefficients (i.e., $|\op{W32.to\_sint}(\texttt{rp}[i])| < 2^{15}$
  for all $i$).  Then executing \jasm{poly\_ntt\_jazz(\texttt{rp})} produces
  an output array $\texttt{rp}'$ satisfying:
  \[
    \op{W32.to\_sint}(\texttt{rp}'[i]) = \op{full\_ntt}(a)[i] \pmod{q},
    \quad i = 0, \dots, 255,
  \]
  where $a[i] = \op{W32.to\_sint}(\texttt{rp}[i])$ for all $i$.
\end{theorem}

\begin{lstlisting}[language=EasyCrypt,
  caption={End-to-end NTT correctness statement},
  label={lst:e2e_ntt}]
lemma jasmin_ntt_full (rp : W32.t Array256.t) :
  (forall i, 0 <= i < 256 =>
     `|W32.to_sint rp.[i]| < 2^15) =>
  hoare[M.poly_ntt_jazz :
    arg = rp ==>
    let a  = Array256.init (fun i => W32.to_sint rp.[i]) in
    let rp' = res in
    forall i, 0 <= i < 256 =>
      W32.to_sint rp'.[i] =
        (NTTFullSpec.full_ntt a).[i] %% q].

lemma haetae_jasmin_ntt_correct (rp : W32.t Array256.t)
    (a : int Array256.t) :
  (forall i, 0 <= i < 256 =>
     W32.to_sint rp.[i] = a.[i] /\ `|a.[i]| < 2^15) =>
  hoare[M.poly_ntt_jazz :
    arg = rp ==>
    let rp' = res in
    forall i, 0 <= i < 256 =>
      W32.to_sint rp'.[i] = (NTTFullSpec.full_ntt a).[i]].
\end{lstlisting}

\subsection{Inverse NTT Correctness}

\begin{theorem}[Jasmin inverse NTT correctness]
  \label{thm:main_invntt}
  Let $\texttt{rp} : \ec{W32.t}^{256}$ encode a polynomial with 23-bit
  coefficients (i.e., $|\op{W32.to\_sint}(\texttt{rp}[i])| < 2^{23}$).
  Then executing \jasm{poly\_invntt\_jazz(\texttt{rp})} produces an output
  array $\texttt{rp}'$ satisfying:
  \[
    \op{W32.to\_sint}(\texttt{rp}'[i])
    \equiv \op{full\_invntt}(a)[i] \cdot R \pmod{q},
    \quad i = 0, \dots, 255,
  \]
  where $a[i] = \op{W32.to\_sint}(\texttt{rp}[i])$ and $R = 2^{32}$.
\end{theorem}

\begin{lstlisting}[language=EasyCrypt,
  caption={End-to-end inverse NTT correctness statement},
  label={lst:e2e_invntt}]
lemma haetae_jasmin_invntt_correct (rp : W32.t Array256.t)
    (a : int Array256.t) :
  (forall i, 0 <= i < 256 =>
     W32.to_sint rp.[i] = a.[i] /\ `|a.[i]| < 2^23) =>
  hoare[M.poly_invntt_jazz :
    arg = rp ==>
    let rp' = res in
    forall i, 0 <= i < 256 =>
      W32.to_sint rp'.[i] =
        ((NTTFullSpec.full_invntt a).[i] * Fq.rmont) %% q].
\end{lstlisting}

The Montgomery factor $\texttt{rmont} = R^{-1} \bmod q = 50386$ in the
postcondition reflects the fact that the inverse NTT output is in Montgomery
form (scaled by $R$).  This is the standard convention in lattice-based
cryptography: NTT outputs are kept in Montgomery form for efficient subsequent
multiplications.

\section{Round-Trip Theorem}

\begin{theorem}[Round-trip correctness]
  \label{thm:roundtrip}
  For any polynomial $a \in \Z_q^{256}$ with $|a[i]| < 2^{15}$:
  \[
    \op{poly\_invntt\_jazz}(\op{poly\_ntt\_jazz}(a))
    \equiv a \cdot R \pmod{q} \text{ (componentwise)}.
  \]
\end{theorem}

\begin{proof}
  Composing \Cref{thm:main_ntt} and \Cref{thm:main_invntt}:
  \begin{align*}
    \op{poly\_ntt\_jazz}(a)[i]
    &= \op{full\_ntt}(a)[i] \pmod{q}, \\
    \op{poly\_invntt\_jazz}(\op{full\_ntt}(a))[i]
    &= \op{full\_invntt}(\op{full\_ntt}(a))[i] \cdot R \pmod{q} \\
    &= a[i] \cdot R \pmod{q},
  \end{align*}
  where the last step uses \Cref{thm:intt_correct}.
\end{proof}

\Cref{thm:roundtrip} is the \emph{operational} correctness statement: applying
the NTT and then the inverse NTT returns the original polynomial (times the
Montgomery factor $R$), which can be absorbed by the Montgomery representation
convention used throughout \haetae{}.

\section{Proof Composition}

The end-to-end proofs are assembled from three chains:

\begin{enumerate}
  \item \textbf{Jasmin $\to$ Imperative}: \Cref{thm:ntt_loop_equiv} from
        \texttt{RefJasminNTT.ec}.
  \item \textbf{Imperative $\to$ Abstract}: \Cref{thm:ntt_correct} from
        \texttt{NTTFullAlgebra.ec}.
  \item \textbf{Abstract $\to$ Round-trip}: \Cref{thm:intt_correct} from
        \texttt{NTTFullSpec.ec}.
\end{enumerate}

The composition is expressed in \easycrypt{} as sequential lemma application:

\begin{lstlisting}[language=EasyCrypt,
  caption={Proof composition in \texttt{NTTEndToEnd.ec}},
  label={lst:e2e_proof}]
proof haetae_jasmin_ntt_correct.
  move => rp a ha.
  (* Step 1: Jasmin -> Imperative (RefJasminNTT) *)
  have h1 := poly_ntt_core_ref ha.
  (* Step 2: Imperative -> Abstract (NTTFullAlgebra) *)
  have h2 := ntt_imp_to_spec h1.
  (* Step 3: Conclude *)
  exact h2.
qed.
\end{lstlisting}

\section{Status of the Proof}

\subsection{Completed Machine-Checked Steps}

\begin{enumerate}
  \item \textbf{\texttt{Fq.ec}}: All lemmas proved, including Montgomery
        reduction correctness, signed arithmetic identities, and bit-width
        bound propagation.

  \item \textbf{\texttt{NTT\_Fq.ec}}: Full imperative specification including
        termination proofs and loop invariants.

  \item \textbf{\texttt{NTTFullSpec.ec}}: Abstract NTT definitions and
        the round-trip identity (pending full discharge of orthogonality
        lemma via the algebra library).

  \item \textbf{\texttt{RefJasminNTT.ec}}: The two main refinement lemmas
        \ec{poly\_ntt\_core\_ref} and \ec{poly\_invntt\_core\_ref}.
\end{enumerate}

\subsection{Remaining Proof Obligation}

The one remaining gap, identified in \texttt{AGENT.md}, is the complete
machine-checked bridge from \ec{NTT\_Fq.NTT.ntt} to \ec{NTTFullSpec.full\_ntt}
via the stage decomposition in \texttt{NTTFullAlgebra.ec}.

Specifically, the butterfly-stage induction lemma (\Cref{lem:butterfly_stage})
requires:
\begin{itemize}
  \item A complete specification of the partial NTT $\NTT_s$ in terms of
        array segments indexed by stage $s$.
  \item A formal connection between the loop counter $k$ and the spectral
        index $\brev_8(2^s + g)$.
  \item The finite sum manipulation proving that the butterfly update
        transforms $\NTT_s$ to $\NTT_{s+1}$.
\end{itemize}

This step is conceptually straightforward but requires significant
proof engineering effort due to the index arithmetic involved.  The
MLKEM development did not provide a reusable template for this exact
structure (\Cref{rem:haetae_mlkem}), requiring the development of new
infrastructure in \texttt{NTTFullAlgebra.ec}.
